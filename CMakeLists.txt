cmake_minimum_required(VERSION 3.20)
project(libpyfairino LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(Threads REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -c "import pathlib,sys,sysconfig; candidates=[pathlib.Path(p)/'cmeel.prefix' for p in sys.path if p]+[pathlib.Path(sysconfig.get_paths()['purelib'])/'cmeel.prefix']; print(next(p for p in candidates if (p/'include/pinocchio/multibody.hpp').exists() and (p/'include/eiquadprog/eiquadprog-rt.hpp').exists()))" OUTPUT_VARIABLE PIN_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT EXISTS "${PIN_PREFIX}/include/pinocchio/multibody.hpp" OR NOT EXISTS "${PIN_PREFIX}/include/eiquadprog/eiquadprog-rt.hpp")
  message(FATAL_ERROR "Install the pinned Pinocchio 4.1 / eiquadprog 1.3.2 build dependencies")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${PIN_PREFIX}")
find_package(pinocchio 4.1.0 EXACT REQUIRED CONFIG)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/official_sdk")
set(IK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/fairino_ik")
add_library(fairino_ik_core STATIC "${IK}/cpp/src/solver.cpp")
target_include_directories(fairino_ik_core PUBLIC "${IK}/cpp/include" "${PIN_PREFIX}/include" /usr/include/eigen3)
target_link_libraries(fairino_ik_core PUBLIC pinocchio::pinocchio_default pinocchio::pinocchio_parsers)
# Keep the vendored source unchanged; adapt only its module entry in the build tree.
file(READ "${IK}/cpp/src/bindings.cpp" IK_BINDINGS)
string(REPLACE "PYBIND11_MODULE(_native, module)" "void bind_kinematics(py::module_ &module)" IK_BINDINGS "${IK_BINDINGS}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ik_bindings.cpp" "${IK_BINDINGS}")
pybind11_add_module(_native NO_EXTRAS src/bindings.cpp src/session.cpp "${CMAKE_CURRENT_BINARY_DIR}/ik_bindings.cpp")
target_include_directories(_native PRIVATE src "${SDK}/include")
target_link_libraries(_native PRIVATE fairino_ik_core "${SDK}/lib/libkine_FR_7Axis.so.1" m Threads::Threads dl)
add_library(_sdk_backend SHARED src/backend.cpp)
target_include_directories(_sdk_backend PRIVATE src "${SDK}/include")
target_link_libraries(_sdk_backend PRIVATE "${SDK}/lib/libhumanorid_sdk.so.3" Threads::Threads)
target_link_options(_sdk_backend PRIVATE "-Wl,-rpath-link,${SDK}/lib" "-Wl,--no-undefined")
set_target_properties(_sdk_backend PROPERTIES PREFIX "")
add_executable(_sdk_worker src/worker.cpp)
target_include_directories(_sdk_worker PRIVATE src)
target_link_libraries(_sdk_worker PRIVATE dl)
foreach(t _native _sdk_backend _sdk_worker)
  set_target_properties(${t} PROPERTIES BUILD_RPATH_USE_ORIGIN TRUE INSTALL_RPATH "$ORIGIN;$ORIGIN/_vendor;$ORIGIN/../cmeel.prefix/lib" INSTALL_RPATH_USE_LINK_PATH FALSE)
endforeach()
# Preserve DT_NEEDED names inside the wheel; no edits to the official ELF files.
install(TARGETS _native _sdk_backend _sdk_worker LIBRARY DESTINATION libpyfairino RUNTIME DESTINATION libpyfairino)
foreach(pair "libhumanorid_sdk.so.3.0.0|libhumanorid_sdk.so.3" "libfairino_motion_planner.so.1.0.4|libfairino_motion_planner.so.1" "libfairino_middleware_logger.so.1.0.0|libfairino_middleware_logger.so.1" "libkine_FR_7Axis.so.1|libkine_FR_7Axis.so")
  string(REPLACE "|" ";" entry "${pair}")
  list(GET entry 0 source)
  list(GET entry 1 dest)
  install(FILES "${SDK}/lib/${source}" DESTINATION libpyfairino/_vendor RENAME "${dest}")
endforeach()
# No SONAME on the kine ELF: use a name-based link to avoid absolute DT_NEEDED.
get_target_property(_libs _native LINK_LIBRARIES)
list(REMOVE_ITEM _libs "${SDK}/lib/libkine_FR_7Axis.so.1")
set_property(TARGET _native PROPERTY LINK_LIBRARIES "${_libs}")
target_link_directories(_native PRIVATE "${SDK}/lib")
target_link_libraries(_native PRIVATE "-l:libkine_FR_7Axis.so")
install(FILES third_party/provenance.json docs/support-matrix.json DESTINATION libpyfairino)
